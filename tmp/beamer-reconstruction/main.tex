\begin{frame}
  \frametitle{The General Heaviside Composite Optimization Problem\\[-1pt]
    {\normalsize\mdseries (G-HSCOP)}}

  \begingroup
  \setlength{\fboxsep}{0.10cm}
  \setlength{\fboxrule}{0.45pt}
  \color{MediumBlack}
  \noindent\fbox{%
    \begin{minipage}{0.94\textwidth}
      \color{MediumBlue}
      \vspace{-0.05cm}
      \noindent\makebox[\linewidth][c]{\(\displaystyle
        \operatorname*{\textbf{maximize}}_{\mathbf{x}\in P}\quad
        \Phi(\mathbf{x})\triangleq c(\mathbf{x})
        +\sum_{k=1}^{K_0}\psi_{0k}(\mathbf{x})
        \underbrace{\mathds{1}_{[0,\infty)}\!\left(\phi_{0k}(\mathbf{x})\right)}_
          {\text{composite indicator fnc.}}
      \)}\par
      \vspace{0.05cm}
      \noindent\makebox[\linewidth][c]{\(\displaystyle
        \operatorname*{\textbf{subject to}}\quad
        \mathbf{A}_{i\bullet}\mathbf{x}
        +\sum_{k=1}^{K_i}\psi_{ik}(\mathbf{x})
        \mathds{1}_{[0,\infty)}\!\left(\phi_{ik}(\mathbf{x})\right)
        \geq \eta_i,\qquad i=1,\ldots,I.
      \)}\par
      \vspace{-0.05cm}
    \end{minipage}%
  }
  \endgroup

  \vspace{0.08cm}
  \footnotesize
  \begin{itemize}
    \setlength{\itemsep}{0.02cm}
    \item $P$ is a \textcolor{MediumRed}{friendly} set, $c$ is a
      \textcolor{MediumRed}{friendly} function, e.g., affine or quadratic;
      \textcolor{MediumRed}{friendly} $=$ amenable to well-known treatments,
      and $\mathbf{A}_{i\bullet}$ is the $i$-th row of a matrix
      $\mathbf{A}\in\mathbb{R}^{I\times n}$.
    \item $\psi_{ik}$ and $\phi_{ik}$ are functions whose properties highlight
      the challenges;
    \item most importantly, $\mathds{1}_{[0,\infty)}$ is the ``closed''
      Heaviside function given by
  \end{itemize}

  \vspace{-0.16cm}
  {\color{MediumBlue}\small
  \[
    \mathds{1}_{[0,\infty)}(s)\triangleq
    \begin{cases}
      1, & \text{if }s\in[0,\infty),\\
      0, & \text{otherwise},
    \end{cases}
    \qquad\text{(upper semi-continuous).}
  \]
  }

\begin{frame}{Conditional Functions and Sparsity}
  \small
  \setlength{\abovedisplayskip}{4pt}
  \setlength{\belowdisplayskip}{4pt}
  \setlength{\abovedisplayshortskip}{3pt}
  \setlength{\belowdisplayshortskip}{3pt}
  The product
  $f(\mathbf{x})\mathds{1}_{[0,\infty)}\!\left(g(\mathbf{x})\right)$
  is a conditional function; i.e., the primary function $f(\mathbf{x})$ is
  activated only if the auxiliary condition $g(\mathbf{x})\geq 0$ holds.

  \vspace{0.12cm}
  {\color{MediumBlue}An immediate example: the sparsity (counting) function}
  \vspace{-0.04cm}
  \[
    |t|_0\triangleq
    \begin{cases}
      1, & \text{if }t\neq 0,\\
      0, & \text{if }t=0,
    \end{cases}
    =\mathds{1}_{(0,\infty)}(|t|)
    =1-\mathds{1}_{[0,\infty)}(-|t|),
  \]
  \vspace{-0.08cm}

  illustrates that the function inside the Heaviside function may not be
  differentiable. This function extends to any step function with a finite
  number of (discontinuous) jumps.

  \vspace{0.14cm}
  An \textcolor{MediumRed}{affine sparsity} constraint
  \vspace{-0.05cm}
  \[
    \sum_{j=1}^{n}a_j|x_j|_0\leq b,
    \qquad \text{with $a_j$ of mixed signs},
  \]
  \vspace{-0.08cm}

  is a non-trivial extension of the cardinality constraint
  $\sum_{j=1}^{n}|x_j|_0\leq K$, whose mixing coefficients are all positive.

  \begin{supplementarynote}
    Positive coefficients preserve lower semicontinuity. With mixed-sign
    $a_j$, upward and downward jumps can coexist, so
    $\sum_{j=1}^{n}a_j|x_j|_0$ may be neither lower nor upper
    semicontinuous.
  \end{supplementarynote}
\end{frame}

\begin{frame}{Bouligand differentiability}
  \small
  \setlength{\abovedisplayskip}{5pt}
  \setlength{\belowdisplayskip}{5pt}

  A function $f:\mathcal{O}\subseteq\mathbb{R}^n\to\mathbb{R}$, where
  $\mathcal{O}$ is an open set in $\mathbb{R}^n$ is
  {\color{MediumBlue}Bouligand differentiable} at
  $\bar{\mathbf{x}}\in\mathcal{O}$ if it is {\color{MediumBlue}locally
  Lipschitz continuous} near $\bar{\mathbf{x}}$ and directionally
  differentiable there with the (finite) directional derivative along the
  direction $\mathbf{v}\in\mathbb{R}^n$ defined by:
  \[
    \fp (\bar{\mathbf{x}};\mathbf{v})\triangleq\lim_{\tau\downarrow0}
    \frac{f(\bar{\mathbf{x}}+\tau\mathbf{v})-f(\bar{\mathbf{x}})}{\tau}.
  \]

  As a function of the direction, $\fp (\bar{\mathbf{x}};\bullet)$ is
  positively homogeneous and globally Lipschitz continuous on $\mathbb{R}^n$
  with the same Lipschitz modulus as that of $f$ near $\bar{\mathbf{x}}$.

  \vspace{0.06cm}
  Calculus rules hold for the directional derivative; these are the sum,
  multiplication, division, and composition rules; in particular,
  \[
    (f\circ g)'(\bar{\mathbf{x}};\mathbf{v})
    =\fp \bigl(g(\bar{\mathbf{x}});g'(\bar{\mathbf{x}};\mathbf{v})\bigr),
  \]

  \textbf{Remark:} the partial differentiation rule may fail;
  \[
    \fp (\bar{\mathbf{x}};\mathbf{v})\neq\sum_{i=1}^{n}
    \fp (\bar{\mathbf{x}};v_i\mathbf{e}^i),\qquad
    \text{where }\ \mathbf{e}^i=i\text{-th coordinate vector}
  \]

  \begin{supplementarynote}
    Convex functions are directionally differentiable.
  \end{supplementarynote}
\end{frame}

\begin{frame}{Piecewise affine functions}
  \small
  \setlength{\abovedisplayskip}{9pt}
  \setlength{\belowdisplayskip}{9pt}

  A {\color{MediumBlue}continuous} function $f:\mathbb{R}^n\to\mathbb{R}^m$
  is {\color{MediumBlue}piecewise affine} if there exists a partition of
  $\mathbb{R}^n$ into a finite number of polyhedra, on each of which, $f$ is
  an affine function.

  \vspace{0.12cm}
  A piecewise affine function is Bouligand differentiable everywhere.
  \textcolor{MediumRed}{Most interested in $m=1$.}

  \vspace{0.12cm}
  A \textcolor{MediumRed}{continuous} function
  $f:\mathbb{R}^n\to\mathbb{R}^m$ is piecewise affine (PA) if and only if
  \[
    f(\mathbf{x})=\underbrace{\max_{1\leq k\leq K}
      \bigl[(\mathbf{a}^k)^{\top}\mathbf{x}+\alpha_k\bigr]}_
      {\text{convex in }\mathbf{x}}
    -\underbrace{\max_{1\leq\ell\leq L}
      \bigl[(\mathbf{b}^{\ell})^{\top}\mathbf{x}+\beta_{\ell}\bigr]}_
      {\text{convex in }\mathbf{x}},\qquad\mathbf{x}\in\mathbb{R}^n
  \]

  for some constant vectors $\mathbf{a}^k$ and $\mathbf{b}^{\ell}$ and
  scalars $\alpha_k$ and $\beta_{\ell}$. Hence, if $f$ is PA, then for every
  $\bar{\mathbf{x}}\in\mathbb{R}^n$,
  \[
    f(\mathbf{x})=f(\bar{\mathbf{x}})
    +\fp (\bar{\mathbf{x}};\mathbf{x}-\bar{\mathbf{x}}),\qquad
    \forall\,\mathbf{x}\text{ sufficiently near }\bar{\mathbf{x}},
  \]
\end{frame}

\begin{frame}{Stationarity conditions}
  \small
  \setlength{\abovedisplayskip}{3pt}
  \setlength{\belowdisplayskip}{3pt}

  Consider an abstract optimization problem
  \[
    \operatorname*{\textbf{minimize}}_{\mathbf{x}\in X}\ f(\mathbf{x}),
    \qquad f\text{ is Bouligand differentiable.}
  \]

  A feasible solution $\bar{\mathbf{x}}\in X$ is a
  {\color{MediumBlue}Bouligand stationary solution} if
  \[
    \boxed{\,\fp (\bar{\mathbf{x}};\mathbf{v})\geq 0\ \text{ for all
      {\color{MediumBlue}tangent vectors} }
      \mathbf{v}\in\mathcal{T}(X;\bar{\mathbf{x}})\,},\quad\text{where}
  \]

  \vspace{-0.06cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.60\textwidth}
    {\color{MediumBlue}$\mathcal{T}(X;\bar{\mathbf{x}})\triangleq$} tangent
    cone of $X$ at $\bar{\mathbf{x}}$
    \[
      \left\{\mathbf{v}\in\mathbb{R}^n\ \middle|\
      \begin{array}{l}
        \mathbf{v}=\displaystyle\lim_{k\to\infty}
          \frac{\mathbf{x}^k-\bar{\mathbf{x}}}{\tau_k}
          \text{ for some}\\[6pt]
        \text{sequences }\{\mathbf{x}^k\}\subset X\text{ converging}\\[2pt]
        \text{to }\bar{\mathbf{x}}\text{ and }\{\tau_k\}\downarrow 0
      \end{array}\right\}
    \]
    \column{0.38\textwidth}
    \textcolor{MediumRed}{\textbf{Two questions:}} How to describe the
    tangent vectors? What if $f$ is not directionally differentiable?
  \end{columns}

  \vspace{0.04cm}
  A feasible solution $\bar{\mathbf{x}}\in X$ is an
  {\color{MediumBlue}epi-stationary solution} if the pair
  $(\bar{t},\bar{\mathbf{x}})$, where $\bar{t}\triangleq f(\bar{\mathbf{x}})$,
  is a Bouligand stationary solution of the lifted problem:
  \[
    \operatorname*{\textbf{minimize}}_{(t,\mathbf{x})\in\mathbb{R}\times X}\
    t\quad\operatorname*{\textbf{subject to}}\ t\geq f(\mathbf{x}).
  \]

  \textcolor{MediumRed}{Note:} if $f$ is Bouligand differentiable, then
  Bouligand stationarity $\Leftrightarrow$ epi-stationarity.
\end{frame}

  \textbf{Two set-theoretic properties:} A set $X\subseteq\mathbb{R}^n$ is
  said to be
  \begin{itemize}
    \setlength{\itemsep}{0.06cm}
    \item {\color{MediumBlue}locally convex-like} near
      $\bar{\mathbf{x}}\in X$ if there exists a neighborhood $\mathcal{N}$ of
      $\bar{\mathbf{x}}$ such that
      $X\cap\mathcal{N}\subseteq\bar{\mathbf{x}}
      +\mathcal{T}(\bar{\mathbf{x}};X)$;
    \item {\color{MediumBlue}locally star-shaped} near
      $\bar{\mathbf{x}}\in X$ if for any $\mathbf{x}\in X$ sufficiently close
      to $\bar{\mathbf{x}}$, there exists a scalar $\bar{\tau}>0$ (dependent
      on $\mathbf{x}$) such that
      $\bar{\mathbf{x}}+\tau(\mathbf{x}-\bar{\mathbf{x}})\in X$ for all
      $\tau\in[0,\bar{\tau}]$.
  \end{itemize}

  \vspace{0.04cm}
  Clearly, convex $\Rightarrow$ locally star-shaped $\Rightarrow$ locally
  convex-like.

  \vspace{0.06cm}
  \begin{itemize}
    \setlength{\itemsep}{0.10cm}
    \item Importantly, if $f$ is Bouligand differentiable and satisfies
      \[
        f(\mathbf{x})\geq f(\bar{\mathbf{x}})
        +\fp (\bar{\mathbf{x}};\mathbf{x}-\bar{\mathbf{x}}),\qquad
        \forall\,\mathbf{x}\text{ sufficiently near }\bar{\mathbf{x}},
        \tag{1}
      \]
      then $\bar{\mathbf{x}}$ is {\color{MediumBlue}Bouligand stationary} on
      a locally convex-like set $X$ if and only if $\bar{\mathbf{x}}$ is a
      \textcolor{MediumRed}{local minimizer} of $f$ on $X$.
    \item Appropriate compositions of convex and PA functions satisfy (1):
      \[
        f=\varphi\circ\Theta\circ\psi,\qquad
        \Theta\triangleq(\theta_{\ell})_{\ell=1}^{L},
      \]
      where $\varphi:\mathbb{R}^L\to\mathbb{R}$ is PA and
      \textcolor{MediumRed}{isotone}, $\theta_{\ell}:\mathbb{R}^m\to\mathbb{R}$
      for $\ell=1,\ldots,L$ is convex, and $\psi:\mathbb{R}^n\to\mathbb{R}^m$
      is PA.
  \end{itemize}
\end{frame}

\begin{frame}{Hierarchical variable selection}
  \small
  \setlength{\abovedisplayskip}{4pt}
  \setlength{\belowdisplayskip}{4pt}
  \setlength{\abovedisplayshortskip}{3pt}
  \setlength{\belowdisplayshortskip}{3pt}

  The statistical model with interaction terms: hypothesizing a bilinear
  relation among the input features
  $\boldsymbol{\xi}=(\xi_i)_{i=1}^{d}$
  \[
    m(\mathbf{x};\boldsymbol{\xi})
    =\sum_{i=1}^{d}x_i^{(1)}\xi_i
    +\sum_{1\leq i<j\leq d}x_{ij}^{(2)}\xi_i\xi_j,
  \]
  with the parameter $\mathbf{x}$ consisting of the unknown model
  coefficients $x_i^{(1)}$ and $x_{ij}^{(2)}$ that obey

  \begin{itemize}
    \setlength{\itemsep}{0.08cm}
    \item {\color{MediumBlue}strong hierarchy}: an interaction term can be
      selected only if both of the linear terms are selected, i.e.,
      \[
        \bigl|x_{ij}^{(2)}\bigr|_0\leq
        \min\bigl\{\bigl|x_i^{(1)}\bigr|_0,\bigl|x_j^{(1)}\bigr|_0\bigr\};
      \]
    \item {\color{MediumBlue}weak hierarchy}: an interaction term can be
      selected only if at least one of the corresponding linear terms is
      selected,
      \[
        \bigl|x_{ij}^{(2)}\bigr|_0\leq
        \bigl|x_i^{(1)}\bigr|_0+\bigl|x_j^{(1)}\bigr|_0;
      \]
  \end{itemize}

  both leading to \textcolor{MediumRed}{mixed-sign} combinations of
  Heaviside functions.
\end{frame}

\begin{frame}{Conditional stochastic functionals}
  \small
  \setlength{\abovedisplayskip}{5pt}
  \setlength{\belowdisplayskip}{5pt}
  \setlength{\abovedisplayshortskip}{4pt}
  \setlength{\belowdisplayshortskip}{4pt}

  \begin{itemize}
    \setlength{\itemsep}{0.12cm}
    \item Expression of a probability function as an expectation:
      \[
        \mathbb{P}\bigl[f(\mathbf{x},\boldsymbol{\xi})\geq 0\bigr]
        =\mathbb{E}\,\mathds{1}_{[0,\infty)}
          \bigl(f(\mathbf{x},\boldsymbol{\xi})\bigr)
        \xrightarrow[\text{approximation (SAA)}]{\text{sample average}}
        \frac{1}{N}\sum_{s=1}^{N}\mathds{1}_{[0,\infty)}
          \bigl(f(\mathbf{x},\boldsymbol{\xi}^s)\bigr);
      \]
    \item and by extension, expression of a
      {\color{MediumBlue}decision-dependent conditional expectation
      function}:
      \[
        \mathbb{E}\bigl[\psi(\mathbf{x},\boldsymbol{\xi})\bigm|
          f(\mathbf{x},\boldsymbol{\xi})\geq 0\bigr]
        \triangleq
        \underbrace{\frac{\mathbb{E}\bigl[\psi(\mathbf{x},\boldsymbol{\xi})
            \mathds{1}_{[0,\infty)}
            (f(\mathbf{x},\boldsymbol{\xi}))\bigr]}
          {\mathbb{P}\bigl[f(\mathbf{x},\boldsymbol{\xi})\geq 0\bigr]}}_
          {\text{fractional Heaviside function}}.
      \]
  \end{itemize}

  \vspace{0.10cm}
  Note the product
  \vspace{-0.02cm}
  \[
    \psi(\mathbf{x},\boldsymbol{\xi})\,
    \mathds{1}_{[0,\infty)}\bigl(f(\mathbf{x},\boldsymbol{\xi})\bigr)
  \]

  raising the possibility of losing semicontinuity (in the decision variable
  $\mathbf{x}$) if the sign of the multiplier function
  $\psi(\bullet,\boldsymbol{\xi})$ is not consistent with the optimization.
\end{frame}

\begin{frame}{Mixed-sign combinations and piecewise inner functions}
  \small
  \setlength{\abovedisplayskip}{4pt}
  \setlength{\belowdisplayskip}{4pt}
  \setlength{\abovedisplayshortskip}{3pt}
  \setlength{\belowdisplayshortskip}{3pt}

  \begin{itemize}
    \setlength{\itemsep}{0.16cm}
    \item First-order stochastic dominance:
      \[
        \mathbb{P}\bigl[f_1(\mathbf{x},\boldsymbol{\xi})\leq c_1\bigr]
        \leq
        \mathbb{P}\bigl[f_2(\mathbf{x},\boldsymbol{\xi})\leq c_2\bigr]
      \]
    \item Conjunctive probability constraint:
      \begin{align*}
        b\;&\geq\;\mathbb{P}\bigl[\psi_1(\mathbf{x},\boldsymbol{\xi})\geq 0
          \ \text{ and }\ \psi_2(\mathbf{x},\boldsymbol{\xi})<0\bigr]\\
        &=\;\mathbb{P}\bigl[\psi_1(\mathbf{x},\boldsymbol{\xi})\geq 0\bigr]
          -\mathbb{P}\bigl[\min\bigl(\psi_1(\mathbf{x},\boldsymbol{\xi}),
          \psi_2(\mathbf{x},\boldsymbol{\xi})\bigr)\geq 0\bigr]
      \end{align*}
    \item Conditional probability constraint:
      \vspace{-0.06cm}
      \[
        b\geq\mathbb{P}\bigl[\psi_1(\mathbf{x},\boldsymbol{\xi})\geq 0
          \bigm|\psi_2(\mathbf{x},\boldsymbol{\xi})\geq 0\bigr]
      \]
      \vspace{-0.42cm}
      \[
        \Updownarrow
      \]
      \vspace{-0.42cm}
      \[
        \mathbb{P}\bigl[\min\bigl(\psi_1(\mathbf{x},\boldsymbol{\xi}),
          \psi_2(\mathbf{x},\boldsymbol{\xi})\bigr)\geq 0\bigr]
        -b\,\mathbb{P}\bigl[\psi_2(\mathbf{x},\boldsymbol{\xi})\geq 0\bigr]
        \leq 0.
      \]
  \end{itemize}
\end{frame}

\begin{frame}

  \begin{itemize}
    \item \textbf{Change-point detection with sparsity.} Let
      $\{\mathbf{x}_t\}_{t=1}^{T}$ be a sequence with
      $\mathbf{x}_t\in\mathbb{R}^p$ and multiple unknown change point
      $\{\tau_i\}_{i=1}^{I}$ with mean shifts
      $\{\boldsymbol{\Delta}_i\}_{i=1}^{I}$ to be determined. Let
      $\boldsymbol{\mu}\in\mathbb{R}^p$ be the baseline mean vector which is
      unknown. This problem is formulated as follows:
  \end{itemize}

  \[
    \begin{aligned}
      \operatorname*{\textbf{minimize}}_{\boldsymbol{\mu},\{\boldsymbol{\Delta}_i\},\{\tau_i\}}\quad
      &\sum_{t=1}^{T}\mathds{1}_{(0,\infty)}(\tau_1-t)
        \bigl\|\mathbf{x}_t-\boldsymbol{\mu}\bigr\|_2^2
        +\sum_{i=1}^{I}\bigl\|\boldsymbol{\Delta}_i\bigr\|_0\\[2pt]
      &+\sum_{i=1}^{I}\sum_{t=1}^{T}
        \boxed{\underbrace{\mathds{1}_{(0,\infty)}(t-\tau_i)\,
          \mathds{1}_{[0,\infty)}(\tau_{i+1}-t)}_
          {\text{expressing }\tau_i<t\leq\tau_{i+1}}}\;
        \bigl\|\mathbf{x}_t-\boldsymbol{\mu}-\boldsymbol{\Delta}_i\bigr\|_2^2
        \\[2pt]
      \operatorname*{\textbf{subject to}}\quad
      &\tau_1\leq\tau_2\leq\cdots\leq\tau_I.
    \end{aligned}
  \]

  \textcolor{MediumRed}{Note} the product of an open and a closed Heaviside
  function.

  \begin{supplementarynote}
    Matched brackets let two factors collapse into one,
    $\mathds{1}_{[0,\infty)}(a)\,\mathds{1}_{[0,\infty)}(b)
     =\mathds{1}_{[0,\infty)}\bigl(\min(a,b)\bigr)$, and likewise for
    $\mathds{1}_{(0,\infty)}$. The mixed pair above does not, since
    $\{a>0,\,b\geq0\}$ is neither open nor closed. Under
    $\tau_i\leq\tau_{i+1}$ it is instead the mixed-sign difference
    $\mathds{1}_{[0,\infty)}(\tau_{i+1}-t)
     -\mathds{1}_{[0,\infty)}(\tau_i-t)$, with affine inner functions.
  \end{supplementarynote}
\end{frame}

\begin{frame}
  \frametitle{Clustering problems:
    {\normalsize\mdseries Classical and extended.}}

  \footnotesize
  \setlength{\abovedisplayskip}{3pt}
  \setlength{\belowdisplayskip}{3pt}
  \setlength{\abovedisplayshortskip}{2pt}
  \setlength{\belowdisplayshortskip}{2pt}

  Let $\Xi\triangleq\{\boldsymbol{\xi}^s\}_{s=1}^{N}$ be a given set of data
  points in $\mathbb{R}^p$. Let
  $\mathbf{c}\triangleq\{\mathbf{c}^k\}_{k=1}^{K}$ be an unknown set of
  cluster centers in $\mathbb{R}^p$. The criterion for a data point
  $\boldsymbol{\xi}^s$ to belong to cluster $k$, written as $s\in k$, is
  \[
    \begin{aligned}
      &\bigl\|\boldsymbol{\xi}^s-\mathbf{c}^k\bigr\|\leq
        \min_{k'\neq k}\bigl\|\boldsymbol{\xi}^s-\mathbf{c}^{k'}\bigr\|\\
      \Leftrightarrow\ \ &h_s(\mathbf{c})\triangleq
        \min_{k'\neq k}\bigl\|\boldsymbol{\xi}^s-\mathbf{c}^{k'}\bigr\|
        -\bigl\|\boldsymbol{\xi}^s-\mathbf{c}^k\bigr\|\geq 0.
    \end{aligned}
  \]

  Define the simple indicator
  \[
    z_{sk}\triangleq\mathds{1}(s\in k)\triangleq
    \mathds{1}_{[0,\infty)}\bigl\{h_s(\mathbf{c})\bigr\}
    =\begin{cases}
      1 & \text{if }s\in k\\
      0 & \text{if }s\notin k
    \end{cases}
  \]

  \vspace{0.06cm}
  {\color{MediumBlue}The classical K-means $(p=2)$ and K-medians $(p=1)$
  problems:}
  \[
    \begin{aligned}
      \operatorname*{\textbf{minimize}}_{\mathbf{c};\,\mathbf{z}}\quad
      &\frac{1}{N}\sum_{s=1}^{N}\Biggl[\,\sum_{k=1}^{K}z_{sk}
        \bigl\|\boldsymbol{\xi}^s-\mathbf{c}^k\bigr\|_p^p\Biggr]\\
      \operatorname*{\textbf{subject to}}\quad
      &\sum_{k=1}^{K}z_{sk}=1,\quad\forall\,s=1,\ldots,N
        &&\begin{array}{@{}l@{}}
            \text{\scriptsize every sample belongs to}\\[-2pt]
            \text{\scriptsize at least one cluster}
          \end{array}\\
      \text{and}\quad
      &z_{sk}\in\{0,1\},\quad\forall\,(s,k)\in[N]\times[K].
    \end{aligned}
  \]
\end{frame}

\begin{frame}
  \frametitle{Clustering problems:
    {\normalsize\mdseries Classical and extended.} (cont.)}

  \footnotesize
  \setlength{\abovedisplayskip}{2pt}
  \setlength{\belowdisplayskip}{2pt}

  Unifying the above two problems is the mixed $\ell_{p,q}$-clustering
  problem with arbitrary pairs of integers $p,q\geq 1$:
  \vspace{-0.22cm}
  \[
    \begin{aligned}
      \operatorname*{\textbf{minimize}}_{\mathbf{c},\mathbf{z}}\quad
      &\underbrace{\frac{1}{N}\sum_{s=1}^{N}\Biggl[\,\sum_{k=1}^{K}z_{sk}
        \bigl\|\boldsymbol{\xi}^s-\mathbf{c}^k\bigr\|_p^p\Biggr]}_
        {\text{bi-convex; denoted WCD}_p(\mathbf{c},\mathbf{z})}
      &&{\color{MediumBlue}\ell_p\text{-norm in objective}}\\[1pt]
      \operatorname*{\textbf{subject to}}\quad
      &\sum_{k=1}^{K}z_{sk}=1,\quad\forall\,s\in[N]\triangleq
        \{1,\ldots,N\}\\
      \text{and}\quad&\text{for all }(s,k)\in[N]\times[K]:\\
      &\underbrace{z_{sk}\leq\mathds{1}_{[0,\infty)}\Bigl(
        \min_{k'\neq k}\bigl\|\boldsymbol{\xi}^s-\mathbf{c}^{k'}\bigr\|_q
        -\bigl\|\boldsymbol{\xi}^s-\mathbf{c}^{k}\bigr\|_q\Bigr)}_
        {\textcolor{MediumRed}{\text{Heaviside constraint with }
          \ell_q\text{-norm}}}\\[1pt]
      &z_{sk}\in\{0,1\},\quad\text{(binary variables)}
    \end{aligned}
  \]

  \textcolor{MediumRed}{Note:} The Heaviside constraint is not needed in the
  classical K-means and K-medians problems where $p=q$.

  \begin{supplementarynote}
    When $p=q$ the objective alone decides membership. Writing the indicator
    out separates the cost ($\ell_p$) from the rule ($\ell_q$), making the
    rule a free modeling choice.
  \end{supplementarynote}
\end{frame}

  $j=1,\ldots,J_i$ and $i=0,\ldots,I$. Given a point
  $\bar{\mathbf{x}}\in\mathbb{R}^n$, if $\exists$ a neighborhood
  $\mathcal{N}$ of $\bar{\mathbf{x}}$ such that for every
  \[
    (i,j)\in\mathcal{J}_0\triangleq\bigl\{(i,j)\mid
      \phi_{ij}(\bar{\mathbf{x}})=0\geq\psi_{ij}(\bar{\mathbf{x}}),\
      i\in[I],\ j\in[J_i]\bigr\},
  \]
  either of the following two conditions hold:
  \begin{itemize}
    \setlength{\itemsep}{0.03cm}
    \item[(A)] for all $\mathbf{x}\in\mathcal{N}$, $\phi_{ij}(\mathbf{x})$ is
      nonnegative if $\psi_{ij}(\bar{\mathbf{x}})<0$,
    \item[(B)] $\psi_{ij}(\mathbf{x})$ is nonnegative for all
      $\mathbf{x}\in\mathcal{N}$;
  \end{itemize}
  then the functions
  \[
    \mathbf{x}\mapsto\sum_{j=1}^{J_i}\psi_{ij}(\mathbf{x})
      \mathds{1}_{[0,\infty)}\bigl(\phi_{ij}(\mathbf{x})\bigr)
  \]
  are upper semicontinuous at $\bar{\mathbf{x}}$ for all $i=0,1,\ldots,I$.
\end{frame}

  For a given $\varepsilon>0$,
  \[
    \begin{aligned}
      \operatorname*{\textbf{maximize}}_{\mathbf{x}\in P}\ \
      &\theta^\varepsilon_{\mathrm{AHS}}\triangleq c(\mathbf{x})
        +\underbrace{\sum_{j=1}^{J_0}\psi_{0j}^+\mathds{1}_{[0,\infty)}
          \bigl(\phi_{0j}(\mathbf{x})\bigr)
          -\sum_{j=1}^{J_0}\psi_{0j}^-\mathds{1}_{(-\varepsilon,\infty)}
          \bigl(\phi_{0j}(\mathbf{x})\bigr)}_{\text{upper semicontinuous}}\\[2pt]
      \operatorname*{\textbf{subject to}}\ \ &\text{for all }i=1,\ldots,I:\\[2pt]
      &\mathbf{A}_{i\bullet}\mathbf{x}
        +\underbrace{\sum_{j=1}^{J_i}\psi_{ij}^+\mathds{1}_{[0,\infty)}
          \bigl(\phi_{ij}(\mathbf{x})\bigr)
          -\sum_{j=1}^{J_i}\psi_{ij}^-\mathds{1}_{(-\varepsilon,\infty)}
          \bigl(\phi_{ij}(\mathbf{x})\bigr)}_{\text{upper semicontinuous}}
        \geq\eta_i,
    \end{aligned}
  \]
  feasible set denoted $X^\varepsilon_{\mathrm{AHS}}$, including $P$:
  \[
    X^{\varepsilon'}_{\mathrm{AHS}}\subseteq X^\varepsilon_{\mathrm{AHS}}
    \subseteq X_{\mathrm{AHS}},\qquad\forall\,\varepsilon'>\varepsilon>0
    \qquad\textcolor{MediumBlue}{[\text{approximating from inside}]}.
  \]
\end{frame}

  \[

  A-HSCOP if and only if $(\bar{\mathbf{x}},\bar{\mathbf{z}})$, where
  $\bar{z}_{ij}\triangleq\mathds{1}_{[0,\infty)}(\phi_{ij}(\bar{\mathbf{x}}))$,

\begin{frame}
  \footnotesize

  {\color{MediumBlue}\textbf{The PIP subproblems.}} Goal is to reduce the
  number of integer variables.

  \vspace{0.04cm}
  For a given scalar $\delta>0$ and vector $\bar{\mathbf{x}}\in
  X_{\mathrm{AHS}}$, let $\mathcal{J}^\delta_{i;>}(\bar{\mathbf{x}})$ and
  $\mathcal{J}^\delta_{i;<}(\bar{\mathbf{x}})$ be two disjoint subsets of
  $\{1,\ldots,J_i\}$ for all $i=0,1,\ldots,I$ such that
  \[
    \begin{aligned}
      \mathcal{J}^\delta_{i;>}(\bar{\mathbf{x}}) &\subseteq
        \underbrace{\bigl\{j\in[J_i]\mid\phi_{ij}(\bar{\mathbf{x}})\geq\delta
          \bigr\}}_{\text{expansion of }\phi_{ij}(\bar{\mathbf{x}})\geq 0}\\[6pt]
      \mathcal{J}^\delta_{i;<}(\bar{\mathbf{x}}) &\subseteq
        \underbrace{\bigl\{j\in[J_i]\mid\phi_{ij}(\bar{\mathbf{x}})\leq-\delta
          \bigr\}}_{\text{expansion of }\phi_{ij}(\bar{\mathbf{x}})\leq 0}
    \end{aligned}\tag{5}
  \]
  and let $\mathcal{J}^\delta_{i;0}(\bar{\mathbf{x}})$ be the complement of
  $\mathcal{J}^\delta_{i;>}(\bar{\mathbf{x}})\cup
  \mathcal{J}^\delta_{i;<}(\bar{\mathbf{x}})$; thus
  \[
    \mathcal{J}^\delta_{i;0}(\bar{\mathbf{x}})\supseteq
      \underbrace{\bigl\{j\in[J_i]\mid -\delta<\phi_{ij}(\bar{\mathbf{x}})
        <\delta\bigr\}}_{\text{indeterminate indices}}
      \supseteq\bigl\{j\in[J_i]\mid\phi_{ij}(\bar{\mathbf{x}})=0\bigr\}.
  \]

  \vspace{0.02cm}
  $\textcolor{MediumRed}{\mathcal{J}^\delta_{i;>}(\bar{\mathbf{x}}),\
    \mathcal{J}^\delta_{i;<}(\bar{\mathbf{x}}),\text{ and }
    \mathcal{J}^\delta_{i;0}(\bar{\mathbf{x}})}$ are
  \textcolor{MediumRed}{working} index sets that determine the binary
  variables to be fixed or solved.
\end{frame}

\begin{frame}
  \footnotesize

  \setlength{\belowdisplayskip}{3pt}

  For a given scalar $\rho>0$, define the
  $\bigl(\mathrm{IP}^{\oplus;\delta}_{\mathrm{AHS};\rho}(\bar{\mathbf{x}})
  \bigr)$:
  \[
    \begin{aligned}
      \operatorname*{\textbf{maximize}}_{\mathbf{x}\in P;\,\mathbf{z}}\ \
      &\theta^{\oplus;\delta}_{\mathrm{AHS};\rho}
        (\mathbf{x},\mathbf{z};\bar{\mathbf{x}})\triangleq c(\mathbf{x})
        +\sum_{j\in\mathcal{J}^\delta_{0;0}(\bar{\mathbf{x}})}\psi_{0j}z_{0j}
        -\underbrace{\frac{\rho}{2}\|\mathbf{x}-\bar{\mathbf{x}}\|_2^2}_{
          \text{proximal term}}\\[2pt]
      \operatorname*{\textbf{subject to}}\ \ &\text{for all }i=1,\ldots,I:\\[1pt]
      &\mathbf{A}_{i\bullet}\mathbf{x}
        +\sum_{j\in\mathcal{J}^\delta_{i;0}(\bar{\mathbf{x}})}\psi_{ij}z_{ij}
        +\sum_{j\in\mathcal{J}^\delta_{i;>}(\bar{\mathbf{x}})}\psi_{ij}
        \geq\eta_i,\\[1pt]
      \textbf{and}\ \ &\text{for all }i=0,1,\ldots,I:\\[1pt]
      &\left.\begin{array}{@{}l@{}}
          \phi_{ij}(\mathbf{x})\geq -M(1-z_{ij})\\[2pt]
          z_{ij}\in\{0,1\}
        \end{array}\right\}\ j\in\mathcal{J}^\delta_{i;0}(\bar{\mathbf{x}})
        \quad\text{\scriptsize integer variables here only}\\[2pt]
      &\phi_{ij}(\mathbf{x})\geq 0\ \text{(equivalent to fixing }z_{ij}=1);
        \ j\in\mathcal{J}^\delta_{i;>}(\bar{\mathbf{x}})\\[1pt]
      &\phi_{ij}(\mathbf{x})\text{ free (equivalent to fixing }z_{ij}=0);
        \ j\in\mathcal{J}^\delta_{i;<}(\bar{\mathbf{x}})
    \end{aligned}
  \]

  \vspace{0.04cm}
  This is a \textcolor{MediumRed}{convex-constrained mixed-binary program},
  solvable by GUROBI.
\end{frame}

\begin{frame}
  \footnotesize
  \setlength{\abovedisplayskip}{3pt}
  \setlength{\belowdisplayskip}{3pt}
  \setlength{\parskip}{0.06cm}

  \textbf{The PIP method for the A-HSCOP} \textcolor{MediumRed}{with
  $\psi_{ij}\geq 0$ and concave $\phi_{ij}$}

  \textbf{Main computation.} Given an iterate $\mathbf{x}^\nu$, select two
  disjoint index sets $\mathcal{J}^\delta_{i;>}(\mathbf{x}^\nu)$ and
  $\mathcal{J}^\delta_{i;<}(\mathbf{x}^\nu)$ satisfying the inclusions in (5).
  Solve the partial $\bigl(\mathrm{IP}^{\oplus;\delta}_{\mathrm{AHS};\rho}
  (\mathbf{x}^\nu)\bigr)$, warm starting at $\mathbf{x}^\nu$.

  \textbf{Termination.} If $\mathbf{x}^\nu$ itself is an optimal solution of
  the self-defined partial IP, stop; the vector $\mathbf{x}^\nu$ is a local
  maximizer of the A-HSCOP.

  \textbf{Updates of index sets.} Otherwise, let $\mathbf{x}^{\nu+1}$ be an
  optimal solution and choose new index sets
  $\mathcal{J}^\delta_{i;>}(\mathbf{x}^{\nu+1})$ and
  $\mathcal{J}^\delta_{i;<}(\mathbf{x}^{\nu+1})$ satisfying (5), ensuring that
  the cardinality
  \[
    \bigl|\mathcal{J}^\delta_{i;>}(\mathbf{x}^{\nu+1})\cup
      \mathcal{J}^\delta_{i;<}(\mathbf{x}^{\nu+1})\bigr|
  \]
  of the new sets is at least one less than the cardinality of the old sets
  at $\mathbf{x}^\nu$. Let $\nu\leftarrow\nu+1$ and repeat the main
  computational step.

  \textcolor{MediumRed}{Note:} (1) The index sets ensure that the iterate
  $\mathbf{x}^{\nu+1}$ will remain feasible to its self-defined IP; and (2)
  expansions of the index sets are subject to variations in implementation.
\end{frame}

\begin{frame}

  \textbf{The case of the A-HSCOP} \textcolor{MediumRed}{with
  $\psi_{ij}\geq 0$ and PA $\phi_{ij}$}
  \[
    \begin{aligned}
      \phi^{\mathrm{PA}}_{ij}(\mathbf{x}) &=
        \underbrace{\max_{1\leq k\leq K_{ij}}\bigl[(\mathbf{a}^k_{ij})^\top
          \mathbf{x}+\alpha^k_{ij}\bigr]}_{
          \text{denoted }\phi^{\mathrm{PAcvx}}_{ij}(\mathbf{x})}
        +\underbrace{\min_{1\leq\ell\leq L_{ij}}\bigl[(\mathbf{b}^\ell_{ij})
          ^\top\mathbf{x}+\beta^\ell_{ij}\bigr]}_{
          \text{denoted }\phi^{\mathrm{PAcve}}_{ij}(\mathbf{x})}\\[2pt]
      &=\max_{1\leq k\leq K_{ij}}\underbrace{\bigl[(\mathbf{a}^k_{ij})^\top
          \mathbf{x}+\alpha^k_{ij}+\phi^{\mathrm{PAcve}}_{ij}(\mathbf{x})
          \bigr]}_{\text{denoted }\phi^{k;+}_{ij}(\mathbf{x}),\text{ each }
          \textcolor{MediumRed}{\text{concave}}}\\[2pt]
      &\geq\phi^{k;+}_{ij}(\mathbf{x})\quad\forall k\in[K_{ij}],
    \end{aligned}
  \]
  for some constant vectors $\mathbf{a}^k_{ij}$ and $\mathbf{b}^\ell_{ij}$
  and scalars $\alpha^k_{ij}$ and $\beta^\ell_{ij}$. For a given vector
  $\bar{\mathbf{x}}$, let for all pairs $(i,j)$,
  \[
    \mathcal{K}_{ij}(\bar{\mathbf{x}})\triangleq\bigl\{k\in[K_{ij}]\mid
      (\mathbf{a}^k_{ij})^\top\bar{\mathbf{x}}+\alpha^k_{ij}\geq
      \phi^{\mathrm{PAcvx}}_{ij}(\bar{\mathbf{x}})\bigr\}.
  \]
  Note that $\phi^{\mathrm{PA}}_{ij}(\bar{\mathbf{x}})
  =\phi^{k;+}_{ij}(\bar{\mathbf{x}})$ for any
  $k\in\widehat{\mathcal{K}}_{ij}\supseteq\mathcal{K}_{ij}(\bar{\mathbf{x}})$,
  in particular for $\widehat{\mathcal{K}}_{ij}$ given by
  \[
    \widehat{\mathcal{K}}^\delta_{ij}(\bar{\mathbf{x}})\triangleq
      \bigl\{k\in[K_{ij}]\mid(\mathbf{a}^k_{ij})^\top\bar{\mathbf{x}}
      +\alpha^k_{ij}\geq\phi^{\mathrm{PAcvx}}_{ij}(\bar{\mathbf{x}})-\delta
      \bigr\},\quad\delta>0.
  \]
\end{frame}

  $\mathbf{k}\triangleq(k_{ij})$:

        \geq\eta_i.

  \vspace{0.02cm}
  feasible set denoted $X^{\oplus;\mathbf{k}}_{\mathrm{AHS}}$; problem denoted
  $\bigl(\mathrm{P}^{\oplus;\mathbf{k}}_{\mathrm{AHS};\rho}\bigr)$

  \vspace{0.06cm}

      \text{for any }\mathbf{x};\quad\text{and}\quad
      X_{\mathrm{AHS}}=\bigcup_{\mathbf{k}}X^{\oplus;\mathbf{k}}_{\mathrm{AHS}}.

\end{frame}

\begin{frame}
  \footnotesize
  \setlength{\abovedisplayskip}{3pt}
  \setlength{\belowdisplayskip}{3pt}

  \textbf{The Iterative Decomposed Shrinkage Algorithm,}
  \textcolor{MediumRed}{IDSA-PIP}, \textbf{for the A-HSCOP}
  \textcolor{MediumRed}{with mixed-signed $\psi_{ij}$ and PA $\phi_{ij}$}

  \vspace{0.04cm}
  Several steps:
  \begin{itemize}
    \setlength{\itemsep}{0.10cm}
    \item Write $\psi_{ij}=\psi^+_{ij}-\psi^-_{ij}$; note
      \[
        -\psi^-_{ij}\mathds{1}_{(-\varepsilon,\infty)}
          \bigl(\phi_{ij}(\mathbf{x})\bigr)
        =-\psi^-_{ij}+\psi^-_{ij}\mathds{1}_{[0,\infty)}
          \bigl(-\phi_{ij}(\mathbf{x})-\varepsilon\bigr)
      \]
    \item Solve a sequence of decomposed $\varepsilon_\nu$-approximation
      problems by the PIP method described above, for a sequence of
      $\{\varepsilon_\nu\}\downarrow 0$.
    \item For each $\varepsilon$-subproblem, need majorization and
      minorization $\phi^{\mathrm{PA}}_{ij}(\mathbf{x})$ by
      \[
        \phi^{\ell;-}_{ij}(\mathbf{x})\leq\phi^{\mathrm{PA}}_{ij}(\mathbf{x})
        \leq\phi^{k;+}_{ij}(\mathbf{x}),
      \]
      respectively.
    \item After majorization/minorization, each PIP subproblem is a reduced
      mixed-binary, concave maximization program (with the only nonlinear
      function being $c(\mathbf{x})$).
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Sketch of IDSA-PIP}
  \small
  \setlength{\abovedisplayskip}{5pt}
  \setlength{\belowdisplayskip}{5pt}
  \setlength{\parskip}{0.14cm}

  \textbf{Initialization.} Let $\{\varepsilon_\nu\}_{\nu=0}^{\infty}\downarrow 0$.
  Let $\mathbf{x}^0\in X^{\varepsilon_0}_{\mathrm{AHS}}$; let
  $(\rho,\delta,\delta')>0$, and $\nu=0$.

  \textbf{Main Computation.} Apply PIP to solve the problem
  $\bigl(\mathrm{P}^{\varepsilon_\nu;k,\ell}_{\mathrm{AHS};\rho}
  (\mathbf{x}^\nu)\bigr)$ for all pairs $(\mathbf{k},\boldsymbol{\ell})$ in
  $\mathcal{M}^\delta(\mathbf{x}^\nu)$ and obtain a local maximizer, denoted
  $\mathbf{x}^{k,\ell}(\mathbf{x}^\nu)$. Determine the maximum of
  \[
    \theta^{\varepsilon_\nu;k;\ell}_{\mathrm{AHS};\rho}
    \bigl(\mathbf{x}^{k,\ell}(\mathbf{x}^\nu);\mathbf{x}^\nu\bigr)
  \]
  among all pairs $(\mathbf{k},\boldsymbol{\ell})$ in
  $\mathcal{M}^\delta(\mathbf{x}^\nu)$.

  \textbf{Termination.} Terminate if $\mathbf{x}^{\nu+1}$ satisfies a prescribed
  rule (e.g., if $\|\mathbf{x}^{\nu+1}-\mathbf{x}^\nu\|$ is within a given
  tolerance). Otherwise, return to the main computation step with
  $\nu\leftarrow\nu+1$.
\end{frame}

  \textcolor{MediumRed}{Every accumulation point} $\mathbf{x}^\infty$ of a
  sequence $\{\mathbf{x}^\nu\}$ produced by the IDSA-PIP algorithm is feasible
  to the \textcolor{MediumRed}{A-HSCOP} (with \textcolor{MediumRed}{mixed
  signed} $\psi_{ij}$).

  \vspace{0.06cm}
  Moreover, if $\mathcal{J}_{i,0}^-(\mathbf{x}^\infty)=\emptyset$ for all
  $i=0,1,\ldots,I$, $\exists$ a pair
  $(\mathbf{k}^\infty,\boldsymbol{\ell}^\infty)\in\mathcal{M}(\mathbf{x}^\infty)$
  such that $\mathbf{x}^\infty$ is a local maximizer of the problem:
  \[
    \begin{aligned}
      \operatorname*{\textbf{maximize}}_{\mathbf{x}\in P}\quad
      &c(\mathbf{x})
        +\sum_{j=1}^{J_0}\psi^+_{0j}\mathds{1}_{[0,\infty)}
          \bigl(\phi^{k^{\infty}_{0j};+}_{0j}(\mathbf{x})\bigr)\\
      &\qquad{}-\sum_{j=1}^{J_0}\psi^-_{0j}\mathds{1}_{[0,\infty)}
          \bigl(\phi^{\ell^{\infty}_{0j};-}_{0j}(\mathbf{x})\bigr)\\[4pt]
      \operatorname*{\textbf{subject to}}\quad

      &\qquad{}-\sum_{j=1}^{J_i}\psi^-_{ij}\mathds{1}_{[0,\infty)}
          \bigl(\phi^{\ell^{\infty}_{ij};-}_{ij}(\mathbf{x})\bigr)
        \geq\eta_i,\quad\text{for all }i\in[I].
    \end{aligned}\tag{6}
  \]

  \vspace{0.04cm}

\begin{frame}
  \footnotesize
  \setlength{\abovedisplayskip}{4pt}
  \setlength{\belowdisplayskip}{4pt}

  \textbf{The multiplicative Heaviside composite problem}
  \textcolor{MediumBlue}{M-HSCOP}:
  \[
    \begin{aligned}
      \operatorname*{\textbf{maximize}}_{\mathbf{x}\in P}\quad
      &c(\mathbf{x})+\sum_{j=1}^{J_0}\psi_{0j}\,
        \mathds{1}_{[0,\infty)}\bigl(\phi_{0j}(\mathbf{x})\bigr)\,
        \mathds{1}_{(0,\infty)}\bigl(\varphi_{0j}(\mathbf{x})\bigr)\\[2pt]

        \mathds{1}_{[0,\infty)}\bigl(\phi_{ij}(\mathbf{x})\bigr)\,
        \mathds{1}_{(0,\infty)}\bigl(\varphi_{ij}(\mathbf{x})\bigr)\geq\eta_i.
    \end{aligned}
  \]

  \vspace{0.04cm}
  \textcolor{MediumBlue}{Approximation of the product:}
  \[
    \psi_{ij}\,
      \underbrace{\mathds{1}_{[0,\infty)}\bigl(\phi_{ij}(\mathbf{x})\bigr)}_{
        \text{closed Heaviside}}\,
      \underbrace{\mathds{1}_{(0,\infty)}\bigl(\varphi_{ij}(\mathbf{x})\bigr)}_{
        \text{open Heaviside}}\ \approx
  \]
  \vspace{-0.30cm}
  \[
    \begin{aligned}
      &\psi^+_{ij}\,\mathds{1}_{[0,\infty)}\bigl(\phi_{ij}(\mathbf{x})\bigr)\,
        \mathds{1}_{[\varepsilon,\infty)}\bigl(\varphi_{ij}(\mathbf{x})\bigr)\\
      &\quad{}-\psi^-_{ij}\,\mathds{1}_{(-\varepsilon,\infty)}
        \bigl(\phi_{ij}(\mathbf{x})\bigr)\,
        \mathds{1}_{(0,\infty)}\bigl(\varphi_{ij}(\mathbf{x})\bigr)
    \end{aligned}
  \]

  \vspace{0.02cm}
  preserves monotonicity and ensures applicability and guaranteed convergence of
  the IDSA-PIP algorithm. (vs.\ an alternative approximation).
\end{frame}

% PDF page 97 -- solving the affine-psi HSCOP: reduction to a mixed-IP.
\begin{frame}
  \footnotesize
  \setlength{\abovedisplayskip}{4pt}
  \setlength{\belowdisplayskip}{4pt}
  \setlength{\parskip}{0.10cm}

  \textcolor{MediumBlue}{Several steps:}

  \textcolor{MediumRed}{Step 1:} Write
  $(\mathbf{a}^{ij})^\top\mathbf{x}+\alpha_{ij}
    =\bigl[(\mathbf{a}^{ij})^\top\mathbf{x}+\alpha_{ij}\bigr]_+
    -\bigl[(\mathbf{a}^{ij})^\top\mathbf{x}+\alpha_{ij}\bigr]_-$,
  \[
    \operatorname*{\textbf{maximize}}_{\mathbf{x}\in P}\ c(\mathbf{x})
    +\sum_{j=1}^{J_0}\left\{
      \begin{aligned}
        &\bigl[(\mathbf{a}^{0j})^\top\mathbf{x}+\alpha_{0j}\bigr]_+
          \mathds{1}_{[0,\infty)}\bigl(\phi_{0j}(\mathbf{x})\bigr)\\
        &{}-\bigl[(\mathbf{a}^{0j})^\top\mathbf{x}+\alpha_{0j}\bigr]_-
          \mathds{1}_{[0,\infty)}\bigl(\phi_{0j}(\mathbf{x})\bigr)
      \end{aligned}\right\}
  \]
  and similarly for constraints.

  \textcolor{MediumRed}{Step 2:} Open up the negative Heaviside terms:
  \[
    \bigl[(\mathbf{a}^{0j})^\top\mathbf{x}+\alpha_{0j}\bigr]_-
      \mathds{1}_{[0,\infty)}\bigl(\phi_{0j}(\mathbf{x})\bigr)
    \approx
    \bigl[(\mathbf{a}^{0j})^\top\mathbf{x}+\alpha_{0j}\bigr]_-
      \mathds{1}_{(-\varepsilon,\infty)}\bigl(\phi_{0j}(\mathbf{x})\bigr).
  \]

  \textcolor{MediumRed}{Step 3:} Linearize the products: $s=yz$, where
  $0\leq y-s\leq\bar{y}(1-z)$ and $0\leq s\leq\bar{y}z$, where
  $0\leq y\leq\bar{y}$ and $z\in\{0,1\}$.

  \textcolor{MediumRed}{Step 4:} Apply IDSA-PIP to the resulting mixed-IP
  formulation.
\end{frame}

\begin{frame}{A-HSCOP with nonnegative coefficients}
  \footnotesize
  \textbf{Assume that:}
  \begin{itemize}
    \setlength{\itemsep}{0.02cm}
    \item \textcolor{MediumRed}{all affine coefficients $\{\psi_{ij}\}$ are
      nonnegative}
    \item affine inner functions $\phi_{ij}(\mathbf{x})$
    \item there exists a constant $\underline{M}$ such that
      $\phi_{ij}(\mathbf{x})\geq -M$ for all $\mathbf{x}\in P$
  \end{itemize}

  \vspace{0.06cm}
  \begin{columns}[t,onlytextwidth]
    \column{0.5\textwidth}
    \textbf{A-HSCOP} $\bigl(\mathrm{P}^{\oplus}_{\mathrm{AHS}}\bigr)$
    \[
      \begin{aligned}
        \operatorname*{\textbf{max}}_{\mathbf{x}\in P}\
        &\theta^{\oplus}_{\mathrm{AHS}}(\mathbf{x})\triangleq c(\mathbf{x})\\
        &\ +\sum_{j=1}^{J_0}\psi_{0j}\mathds{1}_{[0,\infty)}
          \bigl(\phi_{0j}(\mathbf{x})\bigr)\\[2pt]
        \text{s.t.}\
        &\sum_{j=1}^{J_i}\psi_{ij}\mathds{1}_{[0,\infty)}
          \bigl(\phi_{ij}(\mathbf{x})\bigr)\geq b_i,\ \forall i
      \end{aligned}
    \]
    \column{0.5\textwidth}
    \textbf{MIP reformulation}
    \[
      \begin{aligned}
        \operatorname*{\textbf{max}}_{\mathbf{x}\in P;\,\mathbf{z}}\
        &c(\mathbf{x})+\sum_{j=1}^{J_0}\psi_{0j}z_{0j}\\
        \text{s.t.}\
        &\sum_{j=1}^{J_i}\psi_{ij}z_{ij}\geq b_i,\ \forall i\\
        &\phi_{ij}(\mathbf{x})\geq -M(1-z_{ij}),\ \forall j,i\\
        &z_{ij}\in\{0,1\},\ \forall j,i.
      \end{aligned}
    \]
  \end{columns}

  \vspace{0.04cm}
  the number of Heaviside functions $=$ the number of integer variables:\
  $\displaystyle\sum_{i=0}^{m} J_i.$

% Case-studies PDF page 10 -- the PIP algorithm.
\begin{frame}{Computational Algorithm: PIP}
  \small
  \setlength{\parskip}{0.10cm}

  \textbf{Progressive Integer Programming (PIP) algorithm}

  At each iteration with feasible $\bar{\mathbf{x}}$,

  $\bullet$~\textbf{MIP solver:} solve the restricted MIP problem at
  $\bar{\mathbf{x}}$ and obtain an optimal solution $\mathbf{x}_{\mathrm{new}}$
  and corresponding optimal value
  $\theta^{\oplus}_{\mathrm{AHS}}(\mathbf{x}_{\mathrm{new}})$.

  $\bullet$~\textbf{shrink:} if $\theta^{\oplus}_{\mathrm{AHS}}
  (\mathbf{x}_{\mathrm{new}})>\theta^{\oplus}_{\mathrm{AHS}}(\bar{\mathbf{x}})$,
  shrink the in-between interval $[\delta_1,\delta_2]$;

  $\bullet$~\textbf{expand:} if $\theta^{\oplus}_{\mathrm{AHS}}
  (\mathbf{x}_{\mathrm{new}})=\theta^{\oplus}_{\mathrm{AHS}}(\bar{\mathbf{x}})$,
  expand the in-between interval $[\delta_1,\delta_2]$.

  \textbf{Iterative progressive improvement:}
  \[
    \theta^{\oplus}_{\mathrm{AHS}}(\mathbf{x}_{\mathrm{new}})
    \geq\theta^{\oplus}_{\mathrm{AHS}}(\mathbf{x}_{\mathrm{new}};\bar{\mathbf{x}})
    \geq\theta^{\oplus}_{\mathrm{AHS}}(\bar{\mathbf{x}};\bar{\mathbf{x}})
    =\theta^{\oplus}_{\mathrm{AHS}}(\bar{\mathbf{x}})
  \]
  the solution sequence progressively improves, and converges to a local
  maximizer.
\end{frame}

% Case-studies PDF page 11 -- initialization: feasibility via an auxiliary residual variable.
\begin{frame}{Initialization}
  \small
  \setlength{\abovedisplayskip}{4pt}
  \setlength{\belowdisplayskip}{4pt}

  \begin{itemize}
    \setlength{\itemsep}{0.10cm}
    \item It sometimes could be very hard to find a feasible solution satisfying
      a complicated constraint, such as the Gini constraint, or precision
      constraint.
    \item With challenging constraint, one can include
      \textcolor{MediumRed}{an auxiliary variable} as the residual of the
      constraint to guard against infeasibility.
    \item Specifically, we can apply the algorithms to the following problem:
  \end{itemize}

  \[
    \begin{aligned}
      \operatorname*{\textbf{maximize}}_{\mathbf{x}\in P;\,\gamma\geq 0}\quad
      &c(\mathbf{x})+\sum_{j=1}^{J_0}\psi_{0j}\mathds{1}_{[0,\infty)}
        \bigl(\phi_{0j}(\mathbf{x})\bigr)-\lambda\gamma\\
      \operatorname*{\textbf{subject to}}\quad
      &\sum_{j=1}^{J_i}\psi_{ij}\mathds{1}_{[0,\infty)}
        \bigl(\phi_{ij}(\mathbf{x})\bigr)+\gamma\geq\eta_i,\quad i=1,\ldots,I;
    \end{aligned}
  \]

  \noindent where the auxiliary variable $\gamma$ is the constraint residual and
  $\lambda>0$ is a penalty parameter (e.g., $10^4$) whose aim is to enforce the
  feasibility condition $\gamma=0$ if possible.
\end{frame}

% Case-studies PDF page 12 -- initialization: practical PIP start and index-set shrinking.
\begin{frame}{Initialization}
  \small
  \begin{itemize}
    \setlength{\itemsep}{0.12cm}
    \item In practical implementation, one could start with an arbitrary
      solution $\mathbf{x}^0$ with a suitable constant $\gamma_0$ such that the
      constraints are satisfied. For instance, we could let $\mathbf{x}^0$ be an
      unconstrained minimizer of the A-HSCOP together with a suitable $\gamma_0$
      value as the initial feasible solution.
    \item In the first several iterations of PIP, if IP solvers such as
      \textsc{Gurobi} produce a solution with $\gamma=0$, then such a solution
      is also feasible to the original A-HSCOP problem, and we could proceed the
      PIP procedure without the $\gamma$ variable afterwards.
    \item However, it is possible that in the algorithmic process of PIP,
      \textsc{Gurobi} may keep producing solutions with positive $\gamma$
      (infeasible solution) leading to the small progress with very small
      objective values. Possible reason is that for restricted IP subproblems
      with a smaller feasible region or even infeasible by fixing a portion of
      integer variables, it remains challenging for the IP solver to find a
      feasible solution.
    \item Therefore, one effective approach to deal with the above phenomenon is
      to reduce the size of the two fixing index sets
      $\mathcal{J}^\delta_{i;>}(\bar{\mathbf{x}})$ and
      $\mathcal{J}^\delta_{i;<}(\bar{\mathbf{x}})$ until a feasible solution is
      obtained.
  \end{itemize}
\end{frame}

% Case-studies PDF page 13 -- in-between index sets and update rules.
\begin{frame}{In-between Index Sets and Update Rules}
  \footnotesize
  \setlength{\abovedisplayskip}{3pt}
  \setlength{\belowdisplayskip}{3pt}

  \begin{itemize}
    \setlength{\itemsep}{0.05cm}
    \item In principle, after obtaining the new iterate point
      $\mathbf{x}^{\nu+1}$, the algorithm proceed with three updated disjoint
      index sets $\mathcal{J}^\delta_{i;>}(\mathbf{x}^{\nu+1})$,
      $\mathcal{J}^\delta_{i;<}(\mathbf{x}^{\nu+1})$, and
      $\mathcal{J}^\delta_{i;0}(\mathbf{x}^{\nu+1})$, such that the cardinality
      of the in-between index set
      $|\mathcal{J}^\delta_{i;0}(\mathbf{x}^{\nu+1})|$:
      \begin{itemize}
        \setlength{\itemsep}{0cm}
        \item[--] shrink when objective improves
        \item[--] expand when objective does not improve.
      \end{itemize}
    \item How to construct in-between index set
      $\mathcal{J}^\delta_{i;0}(\mathbf{x}^{\nu+1})$, and how it shrinks or
      expands is very critical for practical efficiency. It may depend on the
      application contexts.
  \end{itemize}

  \smallskip
  \hangindent=1.3em\hangafter=1
  1.~(quantile) control the ratio of the integer number to be contained in
  $\mathcal{J}^\delta_{i;0}(\bar{\mathbf{x}})$
  \[
    \mathcal{J}^\delta_{i;0}(\bar{\mathbf{x}})\triangleq
    \bigl\{j\in[J_i]\mid -\delta_{2i}\leq\phi_{ij}(\bar{\mathbf{x}})\leq
    \delta_{1i}\bigr\}
  \]
  where $\delta_{1i},\delta_{2i}$ are r-upper and r-lower quantiles of
  $\{\phi_{ij}(\mathbf{x}^{\nu+1})\}_{j\in[J_i]}$ respectively

  \smallskip
  \hangindent=1.3em\hangafter=1
  2.~(softmax probability) construct the in-between index set via
  \textcolor{MediumBlue}{the softmax-based probabilities} (for clustering
  problem, $\{\xi^s\}$ and $\{\bar{c}^k\}$ represent data points and the
  centroids)
  \[
    \rho^\tau_{sk}\triangleq
    \frac{\exp\{-\|\xi^s-\bar{c}^k\|/\tau\}}
         {\sum_{k=1}^{K}\bigl(\exp\{-\|\xi^s-\bar{c}^k\|/\tau\}\bigr)},
    \qquad
    \mathcal{J}^\tau_0\triangleq
    \bigl\{(s,k):1/K\leq\rho^\tau_{sk}<1/2\bigr\}
  \]

  \smallskip
  $r$ and $\tau$ are \textbf{expanded or shrinked} during PIP iterations.
\end{frame}

% Case-studies PDF page 14 -- in-between index-set update procedure (quantile control).
\begin{frame}{In-between Index Sets and Update Rules}
  \footnotesize
  \setlength{\abovedisplayskip}{2pt}
  \setlength{\belowdisplayskip}{2pt}

  For controlling the ratio of the integer number, we take the following
  procedure:

  \vspace{-0.02cm}
  {\color{MediumBlack}\rule{\linewidth}{0.4pt}}

  \begin{itemize}
    \setlength{\itemsep}{0.03cm}
    \item[--] Compute $\mathbf{x}^{\nu+1}$ an optimal solution of
      $\bigl(\mathrm{P}^{\oplus;\delta}_{\mathrm{AHS}}(\mathbf{x}^\nu)\bigr)$
      \begin{itemize}
        \setlength{\itemsep}{0.01cm}
        \item[--] If $\mathbf{x}^{\nu+1}=\mathbf{x}^\nu$, set
          $r_{\nu+1}\leftarrow\min\{r_\nu+r_\Delta,\,r_{\max}\}$
        \item[--] If $\mathbf{x}^{\nu+1}\neq\mathbf{x}^\nu$, set
          $r_{\nu+1}\leftarrow\max\{r_\nu-r_\Delta,\,r_{\min}\}$
      \end{itemize}
      where $r_\Delta,r_{\max},r_{\min}\in(0,1)$ are predetermined scalars.
    \item[--] Decompose the set of inner function values into two positive and
      non-positive subsets
      \[
        \{\phi_{ij}(\mathbf{x}^{\nu+1})\}_{j\in[J_i]}
        =S^+_{i,\nu+1}\cup S^-_{i,\nu+1}
        =\{\phi^+_{ij}(\mathbf{x}^{\nu+1})\}_{j\in[J_i]}\cup
         \{\phi^-_{ij}(\mathbf{x}^{\nu+1})\}_{j\in[J_i]}
      \]
    \item[--] Compute the lower and upper quantiles
      \[
        \delta^{\nu+1}_{i;1}
          =\text{Lower\_quantile}\bigl(S^+_{i,\nu+1};r_{\nu+1}\bigr),\quad
        \delta^{\nu+1}_{i;2}
          =\text{Upper\_quantile}\bigl(S^-_{i,\nu+1};r_{\nu+1}\bigr)
      \]
  \end{itemize}

  Update in-between index set\quad
  $\mathcal{J}^{\nu+1}_{i;0}\leftarrow
   \bigl\{j\mid\delta^{\nu+1}_{i;2}\leq\phi_{ij}(\mathbf{x}^{\nu+1})\leq
   \delta^{\nu+1}_{i;1}\bigr\}.$

  \vspace{0.02cm}
  {\color{MediumBlack}\rule{\linewidth}{0.4pt}}

  \vspace{0.02cm}
  The aim of the update strategy is to enhance the performance of local
  optimality while saving the computational cost by the shrinkage or truncation
  strategy when needed.
\end{frame}

% Case-studies PDF page 15 -- termination criteria.
\begin{frame}{Termination Criteria}
  \small
  \begin{itemize}
    \setlength{\itemsep}{0.11cm}
    \item In theory, PIP requires to solve a sequence of partial Heaviside
      optimization problems of smaller sizes, which may be solved efficiently by
      IP solvers such as \textsc{Gurobi}.
    \item However, in practice, for the sake of computational efficiency, one
      could embed the early termination strategy when solving the subproblems in
      PIP; for instance, one could use callback mechanism to terminate the
      solver if the objective value remains unchanged for a period of time
      (e.g., 60s).
    \item With early termination of \textsc{Gurobi} in PIP, though the iterate
      point obtained is not guaranteed optimal, the algorithm still has the
      ascent property.
    \item With early termination of \textsc{Gurobi} in PIP, if at the
      terminating iteration, the obtained solution by \textsc{Gurobi} is optimal
      to the partial Heaviside problem, we can still claim that the obtained
      solution has local optimality regarding to the original Heaviside
      optimization problem.
    \item The whole PIP terminates when the objective does not improve for
      several consecutive expansions, yielding a local optimal solution. In
      practical implementation, the number of expansions is usually set
      heuristically depending on the time tolerance.
  \end{itemize}
\end{frame}

% Case-studies PDF page 18 -- A-HSCOP with mixed-signed coefficients and PA inner functions.
% (Case-studies PDF pages 16-17 not photographed.)
\begin{frame}{A-HSCOP with mixed-signed coefficients and PA $\phi_{ij}(\mathbf{x})$}
  \small
  \setlength{\abovedisplayskip}{5pt}
  \setlength{\belowdisplayskip}{5pt}

  \textcolor{MediumBlue}{\textbf{Computational difficulties:}}
  \begin{itemize}
    \setlength{\itemsep}{0.16cm}
    \item $\psi_{ij}=\psi^+_{ij}-\psi^-_{ij}$ are mixed-signed
      ($\psi^+_{ij}\geq 0,\ \psi^-_{ij}\geq 0$), so the objective is not
      upper-semicontinuous and the constrained set is not closed.
    \item The inner function is a general PA $\phi_{ij}(\mathbf{x})$, that is the
      sum of a convex and a concave PA function:
      \[
        \phi_{ij}(\mathbf{x})=\max_{1\leq k\leq K_{ij}}
          \bigl[(\mathbf{a}^k_{ij})^\top\mathbf{x}+\alpha^k_{ij}\bigr]
        +\min_{1\leq\ell\leq L_{ij}}
          \bigl[(\mathbf{b}^\ell_{ij})^\top\mathbf{x}+\beta^\ell_{ij}\bigr]
      \]
  \end{itemize}
\end{frame}

% Case-studies PDF page 19 -- approximation and decomposition for mixed-signed PA A-HSCOP.
% NOTE: the selected-vs-running piece indices (k_{ij}/ell_{ij} vs k'/ell') are a
% careful reading off a dense slide -- flagged for a sanity check.
\begin{frame}{A-HSCOP with mixed-signed coefficients and PA $\phi_{ij}(\mathbf{x})$}
  \footnotesize
  \setlength{\abovedisplayskip}{3pt}
  \setlength{\belowdisplayskip}{3pt}

  \textcolor{MediumBlue}{\textbf{Approximation:}}
  \begin{itemize}
    \setlength{\itemsep}{0.02cm}
    \item With $\psi_{ij}<0$,\
      $\boxed{\psi_{ij}\mathds{1}_{[0,\infty)}\bigl(\phi_{ij}(\mathbf{x})\bigr)
        \ \rightarrow\ \psi_{ij}\mathds{1}_{(-\varepsilon,\infty)}
        \bigl(\phi_{ij}(\mathbf{x})\bigr)}$, which is a lower bound, and upper
      semicontinuous for some $\varepsilon>0$
  \end{itemize}

  \vspace{0.06cm}
  \textcolor{MediumBlue}{\textbf{Decomposition:}}
  \begin{itemize}
    \setlength{\itemsep}{0.02cm}
    \item construct concave PA and convex PA functions with appropriate piece
      selection $(k,\ell)\in\mathcal{M}^\delta(\bar{\mathbf{x}})$ respectively for
      the usage depending on the sign of coefficients:
  \end{itemize}
  \vspace{-0.08cm}
  \[
    \begin{aligned}
      \phi^{k_{ij};+}_{ij}(\mathbf{x}) &\triangleq
        (\mathbf{a}^{k_{ij}}_{ij})^\top\mathbf{x}+\alpha^{k_{ij}}_{ij}
        +\min_{1\leq\ell'\leq L_{ij}}
          \bigl[(\mathbf{b}^{\ell'}_{ij})^\top\mathbf{x}+\beta^{\ell'}_{ij}\bigr],\\
      \phi^{\ell_{ij};-}_{ij}(\mathbf{x}) &\triangleq
        \max_{1\leq k'\leq K_{ij}}
          \bigl[(\mathbf{a}^{k'}_{ij})^\top\mathbf{x}+\alpha^{k'}_{ij}\bigr]
        +(\mathbf{b}^{\ell_{ij}}_{ij})^\top\mathbf{x}+\beta^{\ell_{ij}}_{ij},
    \end{aligned}
  \]

  \vspace{0.02cm}
  where
  \[
    \resizebox{\linewidth}{!}{$\displaystyle
      \mathcal{M}^\delta(\bar{\mathbf{x}})=\prod_{i=0}^{I}\prod_{j=1}^{J_i}
      \Bigl\{\delta\text{-}\operatorname*{argmax}_{1\leq k\leq K_{ij}}
        \bigl[(\mathbf{a}^k_{ij})^\top\bar{\mathbf{x}}+\alpha^k_{ij}\bigr]\Bigr\}
      \times\Bigl\{\delta\text{-}\operatorname*{argmin}_{1\leq\ell\leq L_{ij}}
        \bigl[(\mathbf{b}^\ell_{ij})^\top\bar{\mathbf{x}}+\beta^\ell_{ij}\bigr]\Bigr\}.
    $}
  \]
\end{frame}

\begin{frame}{A-HSCOP with mixed-signed coefficients and PA $\phi_{ij}(\mathbf{x})$}
  \scriptsize
  \setlength{\abovedisplayskip}{3pt}
  \setlength{\belowdisplayskip}{3pt}

  \textcolor{MediumBlue}{Decomposed} $\varepsilon$-approximated subproblem
  $\bigl(\mathrm{P}^{\varepsilon;\mathbf{k},\ell}_{\mathrm{AHS}}\bigr)
  (\bar{\mathbf{x}})$ u.s.c.\ problem
  \[
    \begin{aligned}
      \operatorname*{\textbf{maximize}}_{\mathbf{x}\in P}\
      &V^{\varepsilon;k_0,\ell_0}(\mathbf{x})\triangleq c(\mathbf{x})
        +\sum_{j=1}^{J_0}\psi^+_{0j}\mathds{1}_{[0,\infty)}
          \bigl(\phi^{k_{0j};+}_{0j}(\mathbf{x})\bigr)
        -\sum_{j=1}^{J_0}\psi^-_{0j}\mathds{1}_{(-\varepsilon,\infty)}
          \bigl(\phi^{\ell_{0j};-}_{0j}(\mathbf{x})\bigr)\\
      \operatorname*{\textbf{subject to}}\
      &\sum_{j=1}^{J_i}\psi^+_{ij}\mathds{1}_{[0,\infty)}
          \bigl(\phi^{k_{ij};+}_{ij}(\mathbf{x})\bigr)
        -\sum_{j=1}^{J_i}\psi^-_{ij}\mathds{1}_{(-\varepsilon,\infty)}
          \bigl(\phi^{\ell_{ij};-}_{ij}(\mathbf{x})\bigr)\geq\eta_i,\
          \forall i=1,\ldots,I.
    \end{aligned}
  \]

  \vspace{-0.02cm}
  {\color{MediumBlack}\rule{\linewidth}{0.4pt}}

  For all tuples $(\mathbf{k},\ell)=\bigl\{\{(k_{ij},\ell_{ij})\}_{j=1}^{J_i}
  \bigr\}_{i=0}^{I}$:
  \[
    \begin{aligned}
      &V(\mathbf{x})\geq V^{\varepsilon;k_0,\ell_0}(\mathbf{x})\quad
        \text{for any }\mathbf{x},\\
      &X^{\varepsilon';\mathbf{k},\ell}_{\mathrm{AHS}}\subseteq
        X^{\varepsilon;\mathbf{k},\ell}_{\mathrm{AHS}}\subseteq
        X_{\mathrm{AHS}},\quad\forall\varepsilon'>\varepsilon>0.
    \end{aligned}
  \]
\end{frame}

% Case-studies PDF page 21 -- IDSA-PIP algorithm.
\begin{frame}{Computational Algorithm: IDSA-PIP}
  \small
  \setlength{\parskip}{0.10cm}

  \textbf{Iterative decomposed shrinkage algorithm (IDSA) -- PIP}

  \vspace{-0.04cm}
  {\color{MediumBlack}\rule{\linewidth}{0.4pt}}

  \textbf{Initialization.} Let $\{\varepsilon_\nu\}_{\nu=0}^{\infty}$ be a
  diminishing sequence. Let $\mathbf{x}^0$ be a feasible iterate to the problem
  $\bigl(\mathrm{P}^{\varepsilon_0}_{\mathrm{AHS}}\bigr)$, and set
  $\delta\in\mathbb{R}_+$.

  For $\nu=1,2,\ldots$,

  \textbf{Main Computation.} Solve the problem
  $\bigl(\mathrm{P}^{\varepsilon_\nu;\mathbf{k},\ell}_{\mathrm{AHS}}
  (\mathbf{x}^\nu)\bigr)$ via PIP algorithm for all pairs $(\mathbf{k},\ell)$ in
  $\mathcal{M}^\delta(\mathbf{x}^\nu)$. Set $\mathbf{x}^{\nu+1}$ be the one that
  achieves the maximal objective value among all index pairs.

  \textbf{Terminate} if $\mathbf{x}^{\nu+1}$ satisfies a prescribed rule (e.g.,
  if $\|\mathbf{x}^{\nu+1}-\mathbf{x}^\nu\|$ is within a given tolerance).
\end{frame}

% Case-studies PDF page 22 -- PIP variants for comparison.
\begin{frame}{Computational Algorithm: IDSA-PIP}
  \small
  To evaluate the effectiveness of decomposition and shrinkage techniques, we
  can have various version of PIP for comparisons:
  \begin{enumerate}
    \setlength{\itemsep}{0.12cm}
    \item \textbf{Full MIP}: with fixed $\varepsilon$, solve
      $\bigl(\mathrm{P}^{\varepsilon}_{\mathrm{AHS}}\bigr)$ by the IP solver
      \textsc{Gurobi} to its full MIP formulation with time limit.
    \item \textbf{PIP}: with fixed $\varepsilon$, solve
      $\bigl(\mathrm{P}^{\varepsilon}_{\mathrm{AHS}}\bigr)$ by the PIP method
      (Algorithm 1), without the decomposition of inner PA functions.
    \item \textbf{ISA-PIP}: A variant of the iterative shrinkage method, in
      which the inner loop computation is to apply PIP algorithm to
      $\bigl(\mathrm{P}^{\varepsilon_\nu}_{\mathrm{AHS}}\bigr)$, without the
      decomposition of the inner PA functions.
    \item \textbf{IDSA-PIP}: A variant of the iterative shrinkage method, in
      which the inner loop computation is to apply the PIP method (Algorithm 1)
      to a decomposed approximating subproblem
      $\bigl(\mathrm{P}^{\varepsilon_\nu;\mathbf{k}_\nu,\ell_\nu}_{\mathrm{AHS}}
      \bigr)$ with an index pair $(\mathbf{k}_\nu,\ell_\nu)$
      \textcolor{MediumBlue}{arbitrarily selected} from the active index set for
      the inner PA functions.
  \end{enumerate}
\end{frame}
